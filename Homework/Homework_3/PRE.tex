This assignment develops an elementary approach to the Second Law of Thermodynamics, in the simple context of a Markov Chain with countable state space. It uses martingale theory, and proceeds via reversal of time.

\textbf{Exercise \# 1:} On a probability space $(\Omega, \mathcal{F}, \mathbb{P})$ consider a Markov Chain $Z = \left(Z_n\right)_{n \in \mathbb{N}_0}$ with countable state-space $S$, and time-homogeneous transition probabilities given by a matrix $\Pi = \left(\pi_{xy}\right)_{(x,y) \in S^2}$ with entries
\[
    \pi_{xy} := \mathbb{P}(Z_{n+1} = y \mid Z_n = x), \qquad \forall\, n \in \mathbb{N}_0.
\]
Denote by $\mathbb{F} = \left(\mathcal{F}_n\right)_{n \in \mathbb{N}_0}$ the filtration generated by this Chain, to wit, $\mathcal{F}_n := \sigma(Z_0, Z_1, \cdots, Z_n)$.

Suppose also that the Markov Chain $Z = \left(Z_n\right)_{n \in \mathbb{N}_0}$ is irreducible, positive recurrent, and aperiodic, with initial $\mathbb{P}$-distribution $P^{(0)} = \left(p(0,x)\right)_{x \in S}$ a probability vector with positive components: $p(0,x) = \mathbb{P}(Z_0 = x) > 0$ for all $x \in S$.

We saw last semester that such a Chain has a unique invariant distribution: i.e., a vector $Q = \left(q(y)\right)_{y \in S}$ of positive numbers adding up to 1 and satisfying
\begin{equation}
    q(y) = \sum_{z \in S} q(z)\, \pi_{zy}, \qquad \forall\, y \in S.
\end{equation}
The $k$-step transition probabilities
\begin{equation}
    \pi^{(0)}_{xy} := \mathbf{1}_{x=y}, \qquad \pi^{(k)}_{xy} := \mathbb{P}\!\left(Z_k = y \mid Z_0 = x\right), \quad k \in \mathbb{N}
\end{equation}
converge as $k$ tends to infinity to $q(y)$, for every pair of states $(x,y) \in S^2$; and the finite-dimensional $\mathbb{P}$-distributions of the Chain are given by
\begin{equation}
    \mathbb{P}\!\left(Z_0 = x,\, Z_1 = y_1,\, \cdots,\, Z_n = y_n\right) = p(0,x)\, \pi_{xy_1}\, \pi_{y_1 y_2} \cdots \pi_{y_{n-1} y_n}.
\end{equation}
Denote also by $P^{(n)} = \left(p(n,y)\right)_{y \in S}$ the $\mathbb{P}$-distribution of the random variable $Z_n$: a column vector with components
\[
    p(n,y) := \mathbb{P}(Z_n = y).
\]

Introduce now on the measurable space $(\Omega, \mathcal{F})$ another probability measure $\mathbb{Q}$; under this measure, the initial distribution of the Chain is the vector $Q = \left(q(y)\right)_{y \in S}$ as in (0.1), and thus the finite-dimensional $\mathbb{Q}$-distributions of the Chain are given by
\begin{equation}
    \mathbb{Q}\!\left(Z_0 = x,\, Z_1 = y_1,\, \cdots,\, Z_n = y_n\right) = q(x)\, \pi_{xy_1}\, \pi_{y_1 y_2} \cdots \pi_{y_{n-1} y_n}.
\end{equation}
Then the $\mathbb{Q}$-distribution $Q^{(n)} = \left(q(n,y)\right)_{y \in S}$ of $Z_n$ is $Q^{(n)} \equiv Q$ for all $n \in \mathbb{N}_0$: started at the invariant distribution, the Chain stays with this distribution forever.

\begin{itemize}
    \item[(i)] Argue that the measure $\mathbb{P}$ is absolutely continuous with respect to the measure $\mathbb{Q}$ on $\sigma(Z_n)$ for every $n \in \mathbb{N}_0$, with Radon-Nikod\'ym derivative (``likelihood ratio'')
    \begin{equation}
        \frac{d\mathbb{P}}{d\mathbb{Q}}\bigg|_{\sigma(Z_n)} = \ell\!\left(n, Z_n\right), \quad \text{with} \quad \ell(n,y) := \frac{p(n,y)}{q(y)}, \qquad (n,y) \in \mathbb{N}_0 \times S.
    \end{equation}

    \item[(ii)] Introduce the so-called \textit{relative entropy}
    \[
        H_{P^{(n)}}\!\left(\,\cdot\, \middle|\, Q\right) := \mathbb{E}_{\mathbb{P}}\!\left[\log \ell(n, Z_n)\right]
    \]
    of the distribution $P^{(n)}$ with respect to the distribution $Q$, and argue that, with the function
    \[
        f(u) := u \log(u), \qquad u > 0,
    \]
    this quantity can be cast as
    \begin{equation}
        H_{P^{(n)}}\!\left(\,\cdot\, \middle|\, Q\right) = \mathbb{E}_{\mathbb{Q}}\!\left[\log \ell(n, Z_n) \cdot \frac{d\mathbb{P}}{d\mathbb{Q}}\bigg|_{\sigma(Z_n)}\right] = \mathbb{E}_{\mathbb{Q}}\!\left[f\!\left(\ell(n, Z_n)\right)\right].
    \end{equation}

    \item[(iii)] Fix now an integer $T \in \mathbb{N}$ and consider the time-reversal
    \begin{equation}
        \widehat{Z}_t := Z_{T-t}, \qquad t = 0, 1, \cdots, T
    \end{equation}
    of the Chain over the finite stretch $0, 1, \cdots, T$; as well as the filtration $\widehat{\mathbb{G}} = \left(\widehat{\mathcal{G}}_t\right)_{t=0,1,\cdots,T}$ generated by this time-reversed sequence:
    \[
        \widehat{\mathcal{G}}_t = \sigma\!\left(\widehat{Z}_0, \widehat{Z}_1, \cdots, \widehat{Z}_t\right) = \sigma\!\left(Z_T, Z_{T-1}, \cdots, Z_{T-t}\right).
    \]
    Argue that, under each one of the probability measures $\mathbb{P}$ and $\mathbb{Q}$, the collection of random variables in (0.7) has the Markov property.

    What are the corresponding one-day transition probabilities? are they time-homogeneous?
\end{itemize}

Assume from now on$^1$ also that the state-space $S$ is finite.

\begin{itemize}
    \item[(iv)] With the notation of (0.5), (0.7), show that
    \[
        \left(\ell\!\left(T-t,\, \widehat{Z}_t\right)\right)_{t=0,1,\cdots,T}
    \]
    is a $(\widehat{\mathbb{G}}, \mathbb{Q})$-martingale.

    \item[(v)] Show that the sequence $\mathbb{N}_0 \ni n \mapsto H_{P^{(n)}}\!\left(\,\cdot\,\middle|\,Q\right) \in [0,\infty)$ of relative entropies across time is nonnegative, decreasing, and dissipates down to zero, in the sense
    \[
        \lim_{n \to \infty} \downarrow H_{P^{(n)}}\!\left(\,\cdot\,\middle|\,Q\right) = 0.
    \]

    \item[(vi)] Argue that, for the Total Variation distance
    \[
        \left\|\,\mu - \nu\,\right\|_{\mathrm{TV}} := \sup_{A \in \mathcal{F}} \left|\,\mu(A) - \nu(A)\,\right|
    \]
    between probability measures, we have also
    \[
        \lim_{n \to \infty} \left\|\,P^{(n)} - Q\,\right\|_{\mathrm{TV}} = 0.
    \]
\end{itemize}

\textbf{Remark:} The results in (v), (vi) express of course the Second Law of Thermodynamics in the present context.

\textbf{Remark:} Nothing about the function $f(u) = u\log(u)$, $u > 0$, in (0.6), is exactly ``sacrosanct'' in (iv), (v). We can replace it by any convex function $F : (0,\infty) \to [0,\infty)$ with $F(1) = 0$,$^2$ define the so-called ``$F$-relative entropy''
\[
    H^F\!\left(P^{(n)}\,\middle|\,Q\right) := \mathbb{E}_{\mathbb{P}}\!\left[F\!\left(\frac{\ell(n,Z_n)}{\ell(n,Z_n)}\right)\right] = \mathbb{E}_{\mathbb{Q}}\!\left[F\!\left(\ell(n,Z_n)\right)\right]
\]
in the manner of (ii) above; then everything goes through just as well.

On the other hand, (vi) does need the special form of the original relative entropy with the logarithmic function.

\footnotetext[1]{For simplicity only. The results (iv)--(vi) below continue to hold for countable state-spaces, under the condition $H_{P^{(0)}}(\,\cdot\,|\,Q) < \infty$. Give this a try as well.}
\footnotetext[2]{For instance, $F(u) = u^m - 1$, $u \in (0,\infty)$, for any $m > 1$.}
